% --- Figura de maqueta robusta ---
% Si la imagen existe en figuras/, se inserta como figura con su pie y
% etiqueta. Si la imagen NO existe, no se dibuja nada (ni recuadro ni
% figura vacia); el documento sigue compilando sin errores y el texto
% descriptivo de la pantalla permanece intacto.
% Uso: \maquetafigura{opciones-includegraphics}{archivo}{caption}{label}
\newcommand{\maquetafigura}[4]{%
  \IfFileExists{figuras/#2}{%
    \begin{figure}[H]
      \centering
      \includegraphics[#1]{#2}%
      \caption{#3}\label{#4}%
    \end{figure}%
  }{}%
}
